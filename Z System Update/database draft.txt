CREATE DATABASE IF NOT EXISTS ddms_database;
USE ddms_database;

-- Drop existing tables in correct order (respecting foreign keys)
DROP TABLE IF EXISTS payment_items;
DROP TABLE IF EXISTS payments;
DROP TABLE IF EXISTS dues;
DROP TABLE IF EXISTS students;
DROP TABLE IF EXISTS academic_sessions;
DROP TABLE IF EXISTS programmes;
DROP TABLE IF EXISTS users;
DROP TABLE IF EXISTS roles;
DROP TABLE IF EXISTS audit_logs;

-- 1. ROLES TABLE (Extended with permissions)
CREATE TABLE roles (
    role_id INT PRIMARY KEY AUTO_INCREMENT,
    role_name VARCHAR(50) NOT NULL UNIQUE,
    role_level INT NOT NULL DEFAULT 1,
    can_manage_users BOOLEAN DEFAULT FALSE,
    can_manage_students BOOLEAN DEFAULT FALSE,
    can_process_payments BOOLEAN DEFAULT FALSE,
    can_view_reports BOOLEAN DEFAULT FALSE,
    can_configure_system BOOLEAN DEFAULT FALSE,
    description TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- 2. PROGRAMMES TABLE (Enhanced)
CREATE TABLE programmes (
    programme_id INT PRIMARY KEY AUTO_INCREMENT,
    programme_code VARCHAR(20) NOT NULL UNIQUE,
    programme_name VARCHAR(100) NOT NULL,
    programme_type ENUM('BTech', 'HND', 'Diploma', 'Certificate') NOT NULL,
    duration_years INT DEFAULT 4,
    total_credit_hours INT,
    status ENUM('Active', 'Inactive', 'Discontinued') DEFAULT 'Active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    INDEX idx_programme_status (status),
    INDEX idx_programme_type (programme_type)
);

-- 3. ACADEMIC SESSIONS TABLE
CREATE TABLE academic_sessions (
    session_id INT PRIMARY KEY AUTO_INCREMENT,
    session_name VARCHAR(20) NOT NULL UNIQUE, -- e.g., "2024-2025"
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_current BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY unique_current_session (is_current) WHERE (is_current = TRUE)
);

-- 4. USERS TABLE (Enhanced with more fields)
CREATE TABLE users (
    user_id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(50) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(20),
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    role_id INT NOT NULL,
    profile_image VARCHAR(255),
    last_login TIMESTAMP NULL,
    login_attempts INT DEFAULT 0,
    is_locked BOOLEAN DEFAULT FALSE,
    must_change_password BOOLEAN DEFAULT TRUE,
    is_active BOOLEAN DEFAULT TRUE,
    created_by INT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    FOREIGN KEY (role_id) REFERENCES roles(role_id),
    FOREIGN KEY (created_by) REFERENCES users(user_id),
    INDEX idx_user_status (is_active),
    INDEX idx_user_role (role_id)
);

-- 5. STUDENTS TABLE (Normalized from CSV)
CREATE TABLE students (
    student_id INT PRIMARY KEY AUTO_INCREMENT,
    index_no VARCHAR(20) NOT NULL UNIQUE,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    phone VARCHAR(20),
    programme_id INT NOT NULL,
    programme_level INT NOT NULL CHECK (programme_level BETWEEN 100 AND 500),
    session_type ENUM('Regular', 'Weekend', 'Evening', 'Distance') DEFAULT 'Regular',
    current_academic_year VARCHAR(20),
    position ENUM('student', 'class_rep', 'course_rep', 'executive') DEFAULT 'student',
    status ENUM('Active', 'Inactive', 'Suspended', 'Graduated', 'Withdrawn') DEFAULT 'Active',
    admission_date DATE,
    graduation_date DATE NULL,
    user_id INT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    FOREIGN KEY (programme_id) REFERENCES programmes(programme_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    INDEX idx_student_index (index_no),
    INDEX idx_student_status (status),
    INDEX idx_student_programme (programme_id),
    INDEX idx_student_email (email)
);

-- 6. DUES TABLE (Configurable dues amounts)
CREATE TABLE dues (
    due_id INT PRIMARY KEY AUTO_INCREMENT,
    due_name VARCHAR(100) NOT NULL,
    due_code VARCHAR(50) NOT NULL UNIQUE,
    amount DECIMAL(10,2) NOT NULL,
    currency VARCHAR(3) DEFAULT 'GHS',
    academic_year VARCHAR(20),
    programme_id INT NULL, -- NULL means applicable to all programmes
    is_mandatory BOOLEAN DEFAULT TRUE,
    description TEXT,
    due_date DATE,
    late_fee_percentage DECIMAL(5,2) DEFAULT 0.00,
    status ENUM('Active', 'Inactive', 'Archived') DEFAULT 'Active',
    created_by INT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (programme_id) REFERENCES programmes(programme_id),
    FOREIGN KEY (created_by) REFERENCES users(user_id),
    INDEX idx_due_status (status),
    INDEX idx_due_programme (programme_id)
);

-- 7. PAYMENTS TABLE (Enhanced)
CREATE TABLE payments (
    payment_id INT PRIMARY KEY AUTO_INCREMENT,
    receipt_no VARCHAR(50) NOT NULL UNIQUE,
    student_id INT NOT NULL,
    academic_year VARCHAR(20) NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    amount_paid DECIMAL(10,2) NOT NULL,
    balance DECIMAL(10,2) DEFAULT 0.00,
    payment_date DATE NOT NULL,
    payment_method ENUM('Cash', 'Mobile Money', 'Bank Transfer', 'Cheque', 'Online') DEFAULT 'Cash',
    transaction_id VARCHAR(100),
    payment_status ENUM('Pending', 'Completed', 'Failed', 'Refunded') DEFAULT 'Completed',
    created_by INT NOT NULL,
    verified_by INT NULL,
    verified_at TIMESTAMP NULL,
    notes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (student_id) REFERENCES students(student_id),
    FOREIGN KEY (created_by) REFERENCES users(user_id),
    FOREIGN KEY (verified_by) REFERENCES users(user_id),
    INDEX idx_payment_receipt (receipt_no),
    INDEX idx_payment_student (student_id),
    INDEX idx_payment_date (payment_date),
    INDEX idx_payment_status (payment_status)
);

-- 8. PAYMENT ITEMS TABLE (Line items for each payment)
CREATE TABLE payment_items (
    payment_item_id INT PRIMARY KEY AUTO_INCREMENT,
    payment_id INT NOT NULL,
    due_id INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    academic_year VARCHAR(20),
    description VARCHAR(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (payment_id) REFERENCES payments(payment_id) ON DELETE CASCADE,
    FOREIGN KEY (due_id) REFERENCES dues(due_id),
    INDEX idx_payment_item_payment (payment_id)
);

-- 9. AUDIT LOGS TABLE (For tracking all changes)
CREATE TABLE audit_logs (
    audit_id INT PRIMARY KEY AUTO_INCREMENT,
    table_name VARCHAR(50) NOT NULL,
    record_id INT NOT NULL,
    action ENUM('INSERT', 'UPDATE', 'DELETE', 'LOGIN', 'LOGOUT', 'EXPORT') NOT NULL,
    old_values JSON NULL,
    new_values JSON NULL,
    changed_by INT NULL,
    ip_address VARCHAR(45),
    user_agent TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_audit_table (table_name),
    INDEX idx_audit_action (action),
    INDEX idx_audit_created (created_at),
    FOREIGN KEY (changed_by) REFERENCES users(user_id)
);

-- 10. CONFIGURATION TABLE (System settings)
CREATE TABLE configurations (
    config_id INT PRIMARY KEY AUTO_INCREMENT,
    config_key VARCHAR(100) NOT NULL UNIQUE,
    config_value TEXT,
    config_type ENUM('string', 'integer', 'boolean', 'json', 'array') DEFAULT 'string',
    description TEXT,
    is_public BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Create Views for Reporting
CREATE VIEW student_payment_summary AS
SELECT 
    s.student_id,
    s.index_no,
    s.first_name,
    s.last_name,
    s.email,
    s.phone,
    p.programme_code,
    p.programme_name,
    s.programme_level,
    s.status as student_status,
    COALESCE(SUM(pay.amount_paid), 0) as total_paid,
    COUNT(DISTINCT pay.payment_id) as payment_count,
    MAX(pay.payment_date) as last_payment_date,
    GROUP_CONCAT(DISTINCT pay.receipt_no ORDER BY pay.payment_date DESC SEPARATOR ', ') as receipt_numbers,
    s.created_at
FROM students s
LEFT JOIN programmes p ON s.programme_id = p.programme_id
LEFT JOIN payments pay ON s.student_id = pay.student_id AND pay.payment_status = 'Completed'
WHERE s.deleted_at IS NULL
GROUP BY s.student_id;

CREATE VIEW dues_balance_report AS
SELECT 
    s.student_id,
    s.index_no,
    CONCAT(s.first_name, ' ', s.last_name) as student_name,
    p.programme_name,
    s.academic_year,
    d.due_name,
    d.amount as due_amount,
    COALESCE(SUM(pi.amount), 0) as paid_amount,
    (d.amount - COALESCE(SUM(pi.amount), 0)) as balance
FROM students s
CROSS JOIN dues d
LEFT JOIN payment_items pi ON s.student_id = pi.student_id AND pi.due_id = d.due_id
WHERE d.status = 'Active' AND s.deleted_at IS NULL
GROUP BY s.student_id, d.due_id
HAVING balance > 0;

-- Create Stored Procedures
DELIMITER //

CREATE PROCEDURE GetStudentPaymentHistory(IN student_index VARCHAR(20))
BEGIN
    SELECT 
        p.receipt_no,
        p.payment_date,
        p.total_amount,
        p.payment_method,
        p.payment_status,
        GROUP_CONCAT(CONCAT(d.due_name, ': GHS', pi.amount) SEPARATOR '; ') as items,
        u.username as collected_by
    FROM payments p
    JOIN payment_items pi ON p.payment_id = pi.payment_id
    JOIN dues d ON pi.due_id = d.due_id
    JOIN users u ON p.created_by = u.user_id
    JOIN students s ON p.student_id = s.student_id
    WHERE s.index_no = student_index
    GROUP BY p.payment_id
    ORDER BY p.payment_date DESC;
END//

CREATE PROCEDURE GenerateReceiptNumber()
BEGIN
    DECLARE prefix VARCHAR(10);
    DECLARE year_part VARCHAR(4);
    DECLARE sequence_num INT;
    
    SELECT config_value INTO prefix FROM configurations WHERE config_key = 'receipt_prefix';
    SET year_part = YEAR(CURDATE());
    
    SELECT COALESCE(MAX(CAST(SUBSTRING(receipt_no, LENGTH(prefix) + 5) AS UNSIGNED)), 0) + 1 
    INTO sequence_num 
    FROM payments 
    WHERE receipt_no LIKE CONCAT(prefix, '-', year_part, '%');
    
    SELECT CONCAT(prefix, '-', year_part, '-', LPAD(sequence_num, 6, '0')) as new_receipt_no;
END//

-- Create Triggers for Audit Logging
DELIMITER //

CREATE TRIGGER audit_student_changes
AFTER UPDATE ON students
FOR EACH ROW
BEGIN
    IF OLD.first_name != NEW.first_name OR OLD.last_name != NEW.last_name OR OLD.status != NEW.status THEN
        INSERT INTO audit_logs (table_name, record_id, action, old_values, new_values, changed_by)
        VALUES ('students', NEW.student_id, 'UPDATE',
                JSON_OBJECT('first_name', OLD.first_name, 'last_name', OLD.last_name, 'status', OLD.status),
                JSON_OBJECT('first_name', NEW.first_name, 'last_name', NEW.last_name, 'status', NEW.status),
                NEW.updated_by);
    END IF;
END//

CREATE TRIGGER audit_payment_creation
AFTER INSERT ON payments
FOR EACH ROW
BEGIN
    INSERT INTO audit_logs (table_name, record_id, action, new_values, changed_by)
    VALUES ('payments', NEW.payment_id, 'INSERT',
            JSON_OBJECT('receipt_no', NEW.receipt_no, 'amount', NEW.amount_paid, 'student_id', NEW.student_id),
            NEW.created_by);
END//

DELIMITER ;
